# core/engine.py
import pygame, math
from settings import WIDTH, HEIGHT, TILE

class Camera2D:
    def __init__(self, width, height):
        self.offset = pygame.Vector2(0, 0)
        self.width = width
        self.height = height

    def apply(self, rect):
        return rect.move(-self.offset.x, -self.offset.y)

    def center_on(self, target_rect):
        self.offset.x = target_rect.centerx - WIDTH // 2
        self.offset.y = target_rect.centery - HEIGHT // 2

class Scene:
    def __init__(self, game):
        self.game = game

    def handle_event(self, event): pass
    def update(self, dt): pass
    def draw(self, surface): pass

def draw_text(surf, text, pos, color=(255,255,255), size=20, center=False):
    font = pygame.font.SysFont("arial", size)
    img = font.render(text, True, color)
    rect = img.get_rect()
    if center:
        rect.center = pos
    else:
        rect.topleft = pos
    surf.blit(img, rect)

def clamp(value, minv, maxv):
    return max(minv, min(value, maxv))

def rect_from_grid(x, y, size=TILE):
    return pygame.Rect(x*size, y*size, size, size)

# core/tilemap.py
import pygame, csv
from settings import TILE, GRAY, GREEN, BLUE, YELLOW, PURPLE, ORANGE

COLOR_BY_ID = {
    0: (30, 30, 30),   # floor
    1: (80, 80, 80),   # wall
    2: GREEN,          # grass
    3: BLUE,           # water
    4: YELLOW,         # sand
    5: PURPLE,         # special
    6: ORANGE          # special 2
}

SOLID_IDS = {1, 3}  # walls & water collide

class TileMap:
    def __init__(self, csv_path):
        self.grid = []
        with open(csv_path, newline='') as f:
            reader = csv.reader(f)
            for row in reader:
                self.grid.append([int(x) for x in row])
        self.h = len(self.grid)
        self.w = len(self.grid[0]) if self.h else 0

        # prebuild rects for collisions
        self.solid_rects = []
        for y, row in enumerate(self.grid):
            for x, tid in enumerate(row):
                if tid in SOLID_IDS:
                    self.solid_rects.append(pygame.Rect(x*TILE, y*TILE, TILE, TILE))

    def draw(self, surf, camera):
        # simple fill by color per tile
        for y, row in enumerate(self.grid):
            for x, tid in enumerate(row):
                world_rect = pygame.Rect(x*TILE, y*TILE, TILE, TILE)
                r = camera.apply(world_rect)
                color = COLOR_BY_ID.get(tid, (50, 50, 50))
                pygame.draw.rect(surf, color, r)
                # optional grid lines:
                # pygame.draw.rect(surf, (25,25,25), r, 1)

    def rect_collide(self, rect):
        for r in self.solid_rects:
            if rect.colliderect(r):
                return True
        return False

    def world_size(self):
        return self.w * TILE, self.h * TILE

# game/entities.py
import pygame, random, math
from settings import TILE, PLAYER_SPEED, ENEMY_SPEED, RED, GREEN, BLUE, YELLOW
from core.engine import clamp

def move_with_collision(rect, vel, tilemap, dt):
    # Separate axis collision
    rect.x += vel.x * dt
    if tilemap.rect_collide(rect):
        # move back step by step
        step = 1 if vel.x > 0 else -1
        while tilemap.rect_collide(rect):
            rect.x -= step
    rect.y += vel.y * dt
    if tilemap.rect_collide(rect):
        step = 1 if vel.y > 0 else -1
        while tilemap.rect_collide(rect):
            rect.y -= step
    return rect

class Entity:
    def __init__(self, x, y, w, h, color):
        self.rect = pygame.Rect(x, y, w, h)
        self.color = color
        self.vx = 0
        self.vy = 0
        self.dead = False

    def update(self, dt, tilemap): pass
    def draw(self, surf, camera):
        pygame.draw.rect(surf, self.color, camera.apply(self.rect))

class Player(Entity):
    def __init__(self, x, y):
        super().__init__(x, y, TILE-6, TILE-4, YELLOW)
        self.speed = PLAYER_SPEED
        self.hp = 5
        self.inventory = []  # list of item ids
        self.interact_cooldown = 0.0

    def handle_input(self, keys):
        self.vx = (keys.get("right", False) - keys.get("left", False)) * self.speed
        self.vy = (keys.get("down", False) - keys.get("up", False)) * self.speed

    def update(self, dt, tilemap):
        if self.interact_cooldown > 0:
            self.interact_cooldown -= dt
        vel = pygame.Vector2(self.vx, self.vy)
        self.rect = move_with_collision(self.rect, vel, tilemap, dt)

class NPC(Entity):
    def __init__(self, x, y, name, script_id):
        super().__init__(x, y, TILE-6, TILE-6, (200,200,200))
        self.name = name
        self.script_id = script_id
        self._t = 0.0
        self._dir = random.choice([(1,0),(-1,0),(0,1),(0,-1),(0,0)])

    def update(self, dt, tilemap):
        self._t += dt
        if self._t > 1.5:
            self._t = 0.0
            self._dir = random.choice([(1,0),(-1,0),(0,1),(0,-1),(0,0)])
        vel = pygame.Vector2(self._dir[0]*60, self._dir[1]*60)
        self.rect = move_with_collision(self.rect, vel, tilemap, dt)

class Enemy(Entity):
    def __init__(self, x, y):
        super().__init__(x, y, TILE-6, TILE-6, RED)
        self.speed = ENEMY_SPEED

    def update(self, dt, tilemap, player=None):
        if player:
            dirv = pygame.Vector2(player.rect.center) - pygame.Vector2(self.rect.center)
            if dirv.length_squared() > 1:
                dirv = dirv.normalize() * self.speed
            vel = dirv
        else:
            vel = pygame.Vector2(0,0)
        self.rect = move_with_collision(self.rect, vel, tilemap, dt)

class Item(Entity):
    def __init__(self, x, y, item_id):
        super().__init__(x, y, TILE//2, TILE//2, GREEN)
        self.item_id = item_id

    def picked(self, player):
        player.inventory.append(self.item_id)
        self.dead = True

# game/ui.py
import pygame
from settings import WHITE, BLACK, RED, GREEN

def draw_bar(surface, x, y, w, h, val, max_val, fg, bg):
    pygame.draw.rect(surface, bg, (x, y, w, h))
    ww = int(w * (val / max_val))
    pygame.draw.rect(surface, fg, (x, y, ww, h))
    pygame.draw.rect(surface, (20,20,20), (x, y, w, h), 2)

def draw_hud(surface, player):
    # health
    draw_bar(surface, 20, 20, 200, 18, player.hp, 5, RED, (90,30,30))
    # inventory text
    font = pygame.font.SysFont("arial", 18)
    inv = ", ".join(player.inventory) if player.inventory else "Vacío"
    txt = font.render(f"Inventario: {inv}", True, WHITE)
    surface.blit(txt, (20, 44))

# game/dialogue.py
import json, pygame
from core.engine import draw_text

class DialogueBox:
    def __init__(self, text_lines, who="NPC"):
        self.lines = text_lines
        self.index = 0
        self.who = who
        self.done = False

    def handle_event(self, event):
        if event.type == pygame.KEYDOWN:
            if event.key == pygame.K_SPACE or event.key == pygame.K_RETURN:
                self.index += 1
                if self.index >= len(self.lines):
                    self.done = True

    def draw(self, surf):
        w, h = surf.get_size()
        rect = pygame.Rect(40, h-140, w-80, 100)
        pygame.draw.rect(surf, (0,0,0), rect)
        pygame.draw.rect(surf, (200,200,200), rect, 2)
        if self.index < len(self.lines):
            draw_text(surf, f"{self.who}:", (rect.x+12, rect.y+10), (240,240,240), 20)
            draw_text(surf, self.lines[self.index], (rect.x+12, rect.y+40), (220,220,220), 20)

main.py

python
# main.py
import pygame, sys, json, os
from settings import WIDTH, HEIGHT, FPS, TILE, KEY_UP, KEY_DOWN, KEY_LEFT, KEY_RIGHT, KEY_INTERACT, KEY_INVENTORY
from core.engine import Camera2D, Scene, draw_text
from core.tilemap import TileMap
from game.entities import Player, NPC, Enemy, Item
from game.ui import draw_hud
from game.dialogue import DialogueBox

def keydict():
    keys = pygame.key.get_pressed()
    kd = {
        "up": keys[getattr(pygame, "K_"+KEY_UP)],
        "down": keys[getattr(pygame, "K_"+KEY_DOWN)],
        "left": keys[getattr(pygame, "K_"+KEY_LEFT)],
        "right": keys[getattr(pygame, "K_"+KEY_RIGHT)],
        "interact": keys[getattr(pygame, "K_"+KEY_INTERACT)],
        "inventory": keys[getattr(pygame, "K_"+KEY_INVENTORY)],
    }
    return kd

class PlayScene(Scene):
    def __init__(self, game):
        super().__init__(game)
        # map
        self.map = TileMap(os.path.join("data", "maps", "overworld.csv"))
        # camera
        self.camera = Camera2D(*self.map.world_size())
        # entities
        self.player = Player(3*TILE, 3*TILE)
        self.npcs = [
            NPC(8*TILE, 6*TILE, "Profe", "npc_prof"),
            NPC(30*TILE, 20*TILE, "Guardia", "npc_guard")
        ]
        self.enemies = [
            Enemy(20*TILE, 4*TILE),
            Enemy(34*TILE, 14*TILE)
        ]
        self.items = [
            Item(25*TILE, 8*TILE, "Tarjeta de Acceso")
        ]
        # dialogue
        self.dialogue = None
        with open(os.path.join("data","dialogues.json"), "r", encoding="utf-8") as f:
            self.scripts = json.load(f)
        self.interact_hold = False

    def handle_event(self, event):
        if self.dialogue:
            self.dialogue.handle_event(event)
            if self.dialogue.done:
                self.dialogue = None

    def update(self, dt):
        kd = keydict()
        if not self.dialogue:
            self.player.handle_input(kd)
        self.player.update(dt, self.map)
        for n in self.npcs:
            n.update(dt, self.map)
        for e in self.enemies:
            e.update(dt, self.map, self.player)

        # pick items
        for it in self.items:
            if not it.dead and self.player.rect.colliderect(it.rect):
                it.picked(self.player)
        self.items = [i for i in self.items if not i.dead]

        # interact (E hold protection)
        if kd["interact"] and not self.interact_hold and not self.dialogue:
            self.interact_hold = True
            # check NPC near
            for n in self.npcs:
                if self.player.rect.colliderect(n.rect.inflate(30,30)):
                    lines = self.scripts.get(n.script_id, ["..."])
                    # conditional dialogue with guard
                    if n.script_id == "npc_guard":
                        if "Tarjeta de Acceso" in self.player.inventory:
                            lines = self.scripts.get("npc_congrats", ["Bien."])
                    self.dialogue = DialogueBox(lines, n.name)
                    break
        if not kd["interact"]:
            self.interact_hold = False

        # camera
        self.camera.center_on(self.player.rect)

    def draw(self, surface):
        surface.fill((15,15,20))
        self.map.draw(surface, self.camera)
        # entities
        for it in self.items:
            it.draw(surface, self.camera)
        for n in self.npcs:
            n.draw(surface, self.camera)
        for e in self.enemies:
            e.draw(surface, self.camera)
        self.player.draw(surface, self.camera)
        # UI
        draw_hud(surface, self.player)
        if self.dialogue:
            self.dialogue.draw(surface)

class Game:
    def __init__(self):
        pygame.init()
        pygame.display.set_caption("UP Adventure — Starter Kit")
        self.screen = pygame.display.set_mode((WIDTH, HEIGHT))
        self.clock = pygame.time.Clock()
        self.scene = PlayScene(self)

    def run(self):
        while True:
            dt = self.clock.tick(FPS)/1000.0
            for event in pygame.event.get():
                if event.type == pygame.QUIT:
                    pygame.quit()
                    sys.exit(0)
                self.scene.handle_event(event)
            self.scene.update(dt)
            self.scene.draw(self.screen)
            pygame.display.flip()

if __name__ == "__main__":
    Game().run()
